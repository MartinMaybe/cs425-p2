## Experience

One of the most challenging parts of this project was getting Go-Back-N
retransmission and acknowledgments right when packets were lost, corrupted, 
or duplicated. Being able the sender and receiver recover from damaged traffic 
in those tests and then transfer files successfully through the relay was a
big breakthrough for me. 

Testing everything was really difficult, but seperating the protocol state machines 
from network and file I/O made a huge difference in testing. Copilot was a major
help in creating tests, especially for edge cases like empty files, repeated FIN 
packets, and retransmitting the full unacknowledged window after a timeout.

## Design

The protocol is split into three layers:

- `src/packet.c` encodes and validates wire packets without I/O.
- `src/gbn.c` implements sender and receiver state machines. Events take
  explicit millisecond timestamps and return packet, payload, completion, and
  timer actions.
- `src/file_packets.c` reads files into sequenced DATA packets.
- `src/transfer.c` handles relay registration, UDP, polling, and the monotonic
  clock, and applies actions from the state machines.

## Results

Each run transferred the 1 MiB `1mib.bin` file through `cs425_relay.py
--delay 50` using the default 250 ms retransmission timeout. Every received
copy matched the input with `cmp`. Times are the sender's `time` real values;
throughput is 1024 KiB divided by the mean time.

| Window | Loss | Mean time (s) | Throughput (KiB/s) |
|-------:|-----:|--------------:|-------------------:|
| 1  | 0% | 105.582 | 9.70 |
| 16 | 0% | 6.751   | 151.67 |
| 1  | 5% | 134.152 | 7.63 |
| 16 | 5% | 23.737  | 43.14 |


1. The table’s window-1 no-loss mean is 105.582 seconds. Divide by 1025 packets 
then convert to milliseconds, we get about 103 ms per round trip. If the relay 
adds 100 ms, we get around 3 ms extra. This time comes from the handling of 
packets and ACKs, plus operating-system scheduling.

2. Doing 105.582/6.751 we get that it’s about 15.6 times faster, almost 16. With 
16 packets in flight, the sender can keep transferring while it waits for acknowledgments 
instead of pausing after every packet.

3. With window 1, a timeout resends just the one packet that’s still waiting for 
an ACK. With window 16, a lost packet holds up the receiver’s progress. It discards 
later packets until the gap is filled and the sender has to resend the whole outstanding 
window. That extra retransmission work makes loss hurt window 16 more.

static int parse_unsigned(const char *text, unsigned long maximum,
                          unsigned long *value)
{
  char *end = NULL;
  unsigned long parsed;

  if (text == NULL || text[0] == '\0' || text[0] == '-') {
    return 0;
  }

  errno = 0;
  parsed = strtoul(text, &end, 10);
  if (errno == ERANGE || end == text || *end != '\0' ||
      parsed > maximum) {
    return 0;
  }
  *value = parsed;
  return 1;
}

static int parse_probability(const char *text, double *value)
{
  char *end = NULL;
  double parsed;

  if (text == NULL || text[0] == '\0') {
    return 0;
  }

  errno = 0;
  parsed = strtod(text, &end);
  if (errno == ERANGE || end == text || *end != '\0' ||
      !isfinite(parsed) || parsed < 0.0 || parsed > 0.5) {
    return 0;
  }
  *value = parsed;
  return 1;
}

static int valid_session(const char *session)
{
  size_t length;
  size_t index;

      return 0;
    }
  }
  return 1;

        return invalid(error, error_size,
                       "Port must be an integer from 1 to 65535.");
      }
      options->port = (unsigned int)value;
      break;

file_packets_result file_packets_read(FILE *input, transfer_packet **packets,
                                      size_t *packet_count)
{
  size_t capacity = 0U;
  size_t total_bytes = 0U;

  if (packets == NULL || packet_count == NULL) {
    return FILE_PACKETS_READ_ERROR;
  }
  *packets = NULL;
  *packet_count = 0U;
  if (input == NULL) {
    return FILE_PACKETS_READ_ERROR;
  }

  for (;;) {
    transfer_packet packet;
    size_t count;

    memset(&packet, 0, sizeof(packet));
    packet.type = PACKET_DATA;
    count = fread(packet.payload, 1U, PACKET_MAX_PAYLOAD, input);
    if (count == 0U) {
      if (ferror(input)) { // GCOVR_EXCL_START
        free(*packets);
        *packets = NULL;
        *packet_count = 0U;
        return FILE_PACKETS_READ_ERROR;
      } // GCOVR_EXCL_STOP
      return FILE_PACKETS_OK;
    }
    if (count > (size_t)MAX_FILE_SIZE ||
        total_bytes > (size_t)MAX_FILE_SIZE - count) {
      free(*packets);
      *packets = NULL;
      *packet_count = 0U;
      return FILE_PACKETS_TOO_LARGE;
    }
    total_bytes += count;

    if (*packet_count == capacity) {
      size_t next_capacity = capacity == 0U ? 16U : capacity * 2U;
      transfer_packet *grown;
      grown = realloc(*packets, next_capacity * sizeof(**packets));
      if (grown == NULL) { // GCOVR_EXCL_START
        free(*packets);
        *packets = NULL;
        *packet_count = 0U;
        return FILE_PACKETS_NO_MEMORY;
      } // GCOVR_EXCL_STOP
      *packets = grown;
      capacity = next_capacity;
    }

    packet.seq = (uint32_t)*packet_count;
    packet.length = (uint16_t)count;
    (*packets)[*packet_count] = packet;
    ++(*packet_count);
  }
}

#include "packet.h"

#include <string.h>

static uint16_t read_u16(const unsigned char *bytes)
{
  return (uint16_t)(((uint16_t)bytes[0] << 8U) | (uint16_t)bytes[1]);
}

static uint32_t read_u32(const unsigned char *bytes)
{
  return ((uint32_t)bytes[0] << 24U) |
         ((uint32_t)bytes[1] << 16U) |
         ((uint32_t)bytes[2] << 8U) |
         (uint32_t)bytes[3];
}

static void write_u16(unsigned char *bytes, uint16_t value)
{
  bytes[0] = (unsigned char)(value >> 8U);
  bytes[1] = (unsigned char)(value & 0xffU);
}

static void write_u32(unsigned char *bytes, uint32_t value)
{
  bytes[0] = (unsigned char)(value >> 24U);
  bytes[1] = (unsigned char)((value >> 16U) & 0xffU);
  bytes[2] = (unsigned char)((value >> 8U) & 0xffU);
  bytes[3] = (unsigned char)(value & 0xffU);
}

uint16_t packet_checksum(const unsigned char *bytes, size_t length)
{
  uint32_t sum = 0U;
  size_t offset = 0U;

  while (offset + 1U < length) {
    sum += ((uint32_t)bytes[offset] << 8U) | (uint32_t)bytes[offset + 1U];
    offset += 2U;
  }
  if (offset < length) {
    sum += (uint32_t)bytes[offset] << 8U;
  }
  while ((sum >> 16U) != 0U) {
    sum = (sum & 0xffffU) + (sum >> 16U);
  }
  return (uint16_t)(~sum & 0xffffU);
}

int packet_encode(const transfer_packet *packet, unsigned char *bytes,
                  size_t capacity, size_t *packet_size)
{
  size_t size;
  uint16_t checksum;

  if (packet == NULL || bytes == NULL || packet_size == NULL ||
      (packet->type != PACKET_DATA && packet->type != PACKET_ACK &&
       packet->type != PACKET_FIN) ||
      packet->length > PACKET_MAX_PAYLOAD ||
      (packet->type != PACKET_DATA && packet->length != 0U)) {
    return 0;
  }

  size = PACKET_HEADER_SIZE + (size_t)packet->length;
  if (capacity < size) {
    return 0;
  }

  bytes[0] = (unsigned char)packet->type;
  bytes[1] = 0U;
  bytes[2] = 0U;
  bytes[3] = 0U;
  write_u32(bytes + 4U, packet->seq);
  write_u16(bytes + 8U, packet->length);
  if (packet->length > 0U) {
    memcpy(bytes + PACKET_HEADER_SIZE, packet->payload, packet->length);
  }
  checksum = packet_checksum(bytes, size);
  write_u16(bytes + 2U, checksum);
  *packet_size = size;
  return 1;
}

int packet_decode(const unsigned char *bytes, size_t packet_size,
                  transfer_packet *packet)
{
  uint16_t length;

  if (bytes == NULL || packet == NULL || packet_size < PACKET_HEADER_SIZE ||
      packet_size > PACKET_MAX_SIZE || bytes[0] > (unsigned char)PACKET_FIN ||
      bytes[1] != 0U) {
    return 0;
  }

  length = read_u16(bytes + 8U);
  if (length > PACKET_MAX_PAYLOAD ||
      PACKET_HEADER_SIZE + (size_t)length != packet_size) {
    return 0;
  }
  if (bytes[0] != (unsigned char)PACKET_DATA && length != 0U) {
    return 0;
  }

  if (packet_checksum(bytes, packet_size) != 0U) {
    return 0;
  }

  packet->type = (packet_type)bytes[0];
  packet->seq = read_u32(bytes + 4U);
  packet->length = length;
  if (length > 0U) {
    memcpy(packet->payload, bytes + PACKET_HEADER_SIZE, length);
  }
  return 1;
}

static int64_t monotonic_milliseconds(void)
{
  struct timespec now;

  /* GCOVR_EXCL_START: clock_gettime failure is an OS failure path. */
  if (clock_gettime(CLOCK_MONOTONIC, &now) != 0) {
    return -1;
  }
  /* GCOVR_EXCL_STOP */
  return ((int64_t)now.tv_sec * INT64_C(1000)) +
         ((int64_t)now.tv_nsec / INT64_C(1000000));
}

static int open_relay_socket(const cli_options *options)
{
  const struct addrinfo *address;
  int fd;

  for (address = options->relay_addresses; address != NULL;
       address = address->ai_next) {
    fd = socket(address->ai_family, address->ai_socktype, address->ai_protocol);
    /* GCOVR_EXCL_START: socket creation failure is an OS failure path. */
    if (fd < 0) {
      continue;
    }
    /* GCOVR_EXCL_STOP */
    if (connect(fd, address->ai_addr, address->ai_addrlen) == 0) {
      return fd;
    }
    (void)close(fd);
  }
  (void)fprintf(stderr, "Unable to connect UDP socket to relay: %s\n",
                strerror(errno));
  return -1;
}

static int wait_readable(int fd, int timeout_ms)
{
  struct pollfd descriptor;
  int result;
  int64_t start = 0;
  int64_t deadline = 0;

  descriptor.fd = fd;
  descriptor.events = POLLIN;
  descriptor.revents = 0;
  if (timeout_ms > 0) {
    start = monotonic_milliseconds();
    /* GCOVR_EXCL_START: monotonic clock failure is an OS failure path. */
    if (start < 0) {
      return -1;
    }
    /* GCOVR_EXCL_STOP */
    deadline = start + timeout_ms;
  }
  for (;;) {
    result = poll(&descriptor, 1U, timeout_ms);
    /* GCOVR_EXCL_START: poll errors and EINTR are OS failure paths. */
    if (result >= 0 || errno != EINTR) {
      break;
    }
    /* GCOVR_EXCL_STOP */
    /* GCOVR_EXCL_START: timeout-zero after poll EINTR is an OS edge case. */
    if (timeout_ms == 0) {
      return 0;
    }
    /* GCOVR_EXCL_STOP */
    /* GCOVR_EXCL_START: this clock/deadline adjustment only follows poll
       returning EINTR. */
    {
      int64_t now = monotonic_milliseconds();
      int64_t remaining;
      if (now < 0) {
        return -1;
      }
      remaining = deadline - now;
      if (remaining <= 0) {
        return 0;
      }
      timeout_ms = (int)remaining;
    }
    /* GCOVR_EXCL_STOP */
  }
  /* GCOVR_EXCL_START: poll error flags are reported by the OS. */
  if (result > 0 &&
      (descriptor.revents & (POLLERR | POLLHUP | POLLNVAL)) != 0) {
    errno = EIO;
    return -1;
  }
  /* GCOVR_EXCL_STOP */
  return result;
}

static int register_receiver(int fd, const char *session)
{
  char hello[48];
  char reply[256];
  int hello_length;
  int attempt;

  hello_length = snprintf(hello, sizeof(hello), "HELLO %s recv", session);
  if (hello_length < 0 || (size_t)hello_length >= sizeof(hello)) {
    (void)fprintf(stderr, "Could not construct receiver registration.\n");
    return 0;
  }

  for (attempt = 0; attempt < REGISTRATION_ATTEMPTS; ++attempt) {
    int ready;
    ssize_t sent = send(fd, hello, (size_t)hello_length, 0);
    /* GCOVR_EXCL_START: send failure is an OS failure path. */
    if (sent != hello_length) {
      (void)fprintf(stderr, "Could not send registration to relay: %s\n",
                    sent < 0 ? strerror(errno) : "short UDP datagram");
      return 0;
    }
    /* GCOVR_EXCL_STOP */

    ready = wait_readable(fd, REGISTRATION_TIMEOUT_MS);
    /* GCOVR_EXCL_START: poll failure is an OS failure path. */
    if (ready < 0) {
      (void)fprintf(stderr, "Error waiting for relay registration: %s\n",
                    strerror(errno));
      return 0;
    }
    /* GCOVR_EXCL_STOP */
    if (ready == 0) {
      continue;
    }

    {
      ssize_t received = recv(fd, reply, sizeof(reply) - 1U, 0);
      /* GCOVR_EXCL_START: recv failures are OS failure paths. */
      if (received < 0) {
        if (errno == EINTR) {
          --attempt;
          continue;
        }
        (void)fprintf(stderr, "Could not receive relay registration: %s\n",
                      strerror(errno));
        return 0;
      }
      /* GCOVR_EXCL_STOP */
      reply[received] = '\0';
      if (received == 2 && memcmp(reply, "OK", 2U) == 0) {
        return 1;
      }
      if (received >= 4 && memcmp(reply, "ERR ", 4U) == 0) {
        (void)fprintf(stderr, "Relay refused registration: %s\n", reply + 4);
        return 0;
      }
      (void)fprintf(stderr, "Unexpected relay registration reply: %s\n", reply);
      return 0;
    }
  }

  (void)fprintf(stderr, "Relay registration timed out after %d attempts.\n",
                REGISTRATION_ATTEMPTS);
  return 0;
}

static int send_protocol_packets(int fd, const gbn_actions *actions)
{
  size_t index;

  for (index = 0U; index < actions->packet_count; ++index) {
    unsigned char bytes[PACKET_MAX_SIZE];
    size_t packet_size;
    ssize_t sent;

    /* GCOVR_EXCL_START: packet encoding failure violates state output
       invariants; send failure is an OS failure path. */
    if (!packet_encode(&actions->packets[index], bytes, sizeof(bytes),
                       &packet_size)) {
      (void)fprintf(stderr, "Could not encode protocol packet.\n");
      return 0;
    }
    sent = send(fd, bytes, packet_size, 0);
    if (sent != (ssize_t)packet_size) {
      (void)fprintf(stderr, "Could not send packet to relay: %s\n",
                    sent < 0 ? strerror(errno) : "short UDP datagram");
      return 0;
    }
    /* GCOVR_EXCL_STOP */
  }
  return 1;
}

static int receive_packets(int fd, FILE **output, gbn_receiver *receiver)
{
  unsigned char bytes[PACKET_MAX_SIZE + 1U];
  transfer_packet packet;
  gbn_actions actions;

  for (;;) {
    int64_t now = monotonic_milliseconds();
    int ready;
    ssize_t received;

    /* GCOVR_EXCL_START: monotonic clock failure is an OS failure path. */
    if (now < 0) {
      (void)fprintf(stderr, "Could not read monotonic clock: %s\n",
                    strerror(errno));
      return 0;
    }
    /* GCOVR_EXCL_STOP */
    {
      int64_t deadline = receiver->phase == GBN_RECEIVER_ACTIVE
                             ? receiver->idle_deadline_ms
                             : receiver->linger_deadline_ms;
      int64_t remaining = deadline - now;
      ready = wait_readable(
          fd, remaining > (int64_t)INT_MAX ? INT_MAX : (int)remaining);
    }
    /* GCOVR_EXCL_START: poll failure is an OS failure path. */
    if (ready < 0) {
      (void)fprintf(stderr, "Network error while receiving packet: %s\n",
                    strerror(errno));
      return 0;
    }
    /* GCOVR_EXCL_STOP */
    if (ready == 0) {
      now = monotonic_milliseconds();
      /* GCOVR_EXCL_START: monotonic clock failure is an OS failure path. */
      if (now < 0) {
        (void)fprintf(stderr, "Could not read monotonic clock: %s\n",
                      strerror(errno));
        return 0;
      }
      /* GCOVR_EXCL_STOP */
      gbn_receiver_on_timeout(receiver, now, &actions);
      if (actions.completed) {
        return 1;
      }
      if (actions.failed) {
        (void)fprintf(stderr,
                      "Receiver timed out waiting for valid packets.\n");
        return 0;
      }
      continue; /* GCOVR_EXCL_LINE: poll's timeout boundary is OS-scheduled. */
    } else { /* GCOVR_EXCL_LINE: branch is determined by poll readiness. */
      /* GCOVR_EXCL_START: this boundary race is determined by whether poll
         reports a queued datagram at the exact monotonic idle deadline. */
      if (now >= receiver->idle_deadline_ms) {
        gbn_receiver_on_timeout(receiver, now, &actions);
        (void)fprintf(stderr,
                      "Receiver timed out waiting for valid packets.\n");
        return 0;
      }
      /* GCOVR_EXCL_STOP */

      received = recv(fd, bytes, sizeof(bytes), 0);
      /* GCOVR_EXCL_START: recv failures are OS failure paths. */
      if (received < 0) {
        if (errno == EINTR) {
          continue;
        }
        (void)fprintf(stderr, "Network error while receiving packet: %s\n",
                      strerror(errno));
        return 0;
      }
      /* GCOVR_EXCL_STOP */
      if ((size_t)received > PACKET_MAX_SIZE ||
          !packet_decode(bytes, (size_t)received, &packet)) {
        continue;
      }
      now = monotonic_milliseconds();
      /* GCOVR_EXCL_START: monotonic clock failure is an OS failure path. */
      if (now < 0) {
        (void)fprintf(stderr, "Could not read monotonic clock: %s\n",
                      strerror(errno));
        return 0;
      }
      /* GCOVR_EXCL_STOP */

      gbn_receiver_on_packet(receiver, &packet, now, &actions);
      if (actions.deliver_payload && actions.payload_length > 0U &&
          fwrite(actions.payload, 1U, actions.payload_length, *output) !=
              (size_t)actions.payload_length) {
        /* GCOVR_EXCL_START: fwrite failure is a standard-library failure path.
         */
        (void)fprintf(stderr, "Could not write received file: %s\n",
                      strerror(errno));
        return 0;
        /* GCOVR_EXCL_STOP */
      }
      if (actions.finish_output) {
        /* GCOVR_EXCL_START: fclose failure is a standard-library failure path.
         */
        if (fclose(*output) != 0) {
          *output = NULL;
          (void)fprintf(stderr, "Could not close received file: %s\n",
                        strerror(errno));
          return 0;
        }
        /* GCOVR_EXCL_STOP */
        *output = NULL;
      }
      /* GCOVR_EXCL_START: protocol output failures are covered by the packet
         encoder/send system-call exclusions above. */
      if (!send_protocol_packets(fd, &actions)) {
        return 0;
      }
      /* GCOVR_EXCL_STOP */
    }
  }
}

static int load_packets(const char *path, packet_list *list)
{
  FILE *input;
  file_packets_result result;

  memset(list, 0, sizeof(*list));
  input = fopen(path, "rb");
  /* GCOVR_EXCL_START: fopen failure is a standard-library failure path. */
  if (input == NULL) {
    (void)fprintf(stderr, "Could not open input file '%s': %s\n", path,
                  strerror(errno));
    return 0;
  }
  /* GCOVR_EXCL_STOP */

  result = file_packets_read(input, &list->packets, &list->count);
  /* GCOVR_EXCL_START: fclose failure is a standard-library failure path. */
  if (fclose(input) != 0) {
    (void)fprintf(stderr, "Could not close input file '%s': %s\n", path,
                  strerror(errno));
    free(list->packets);
    memset(list, 0, sizeof(*list));
    return 0;
  }
  /* GCOVR_EXCL_STOP */
  if (result != FILE_PACKETS_OK) {
    if (result == FILE_PACKETS_TOO_LARGE) {
      (void)fprintf(stderr, "Input file exceeds the 16 MiB transfer limit.\n");
      /* GCOVR_EXCL_START: packet allocation failure is a library failure path.
       */
    } else if (result == FILE_PACKETS_NO_MEMORY) {
      (void)fprintf(stderr, "Could not allocate packet buffer.\n");
      /* GCOVR_EXCL_STOP */
      /* GCOVR_EXCL_START: fread error is a standard-library failure path. */
    } else {
      (void)fprintf(stderr, "Could not read input file '%s'.\n", path);
      /* GCOVR_EXCL_STOP */
    }
    free(list->packets);
    memset(list, 0, sizeof(*list));
    return 0;
  }
  return 1;
}

int send_transfer(const cli_options *options)
{
  packet_list packets;
  gbn_sender sender;
  gbn_actions actions;
  int fd = -1;
  int64_t now;
  int success = 0;

  if (!load_packets(options->file, &packets)) {
    return 2;
  }
  if (!gbn_sender_init(&sender, packets.packets, (uint32_t)packets.count,
                       options->window, (int64_t)options->timeout_ms)) {
    (void)fprintf(stderr, "Could not initialize sender state machine.\n");
    free(packets.packets);
    return 2;
  }

  fd = open_relay_socket(options);
  if (fd < 0) {
    free(packets.packets);
    return 2;
  }
  if (!register_sender(fd, options)) {
    (void)close(fd);
    free(packets.packets);
    return 2;
  }

  now = monotonic_milliseconds();
  /* GCOVR_EXCL_START: monotonic clock failure is an OS failure path. */
  if (now < 0) {
    (void)fprintf(stderr, "Could not read monotonic clock: %s\n",
                  strerror(errno));
    goto done;
  }
  /* GCOVR_EXCL_STOP */
  gbn_sender_start(&sender, now, &actions);
  /* GCOVR_EXCL_START: packet encoding/send failure comes from library/OS. */
  if (!send_protocol_packets(fd, &actions)) {
    goto done;
  }
  /* GCOVR_EXCL_STOP */

  while (!sender.completed && !sender.failed) {
    int ready;
    now = monotonic_milliseconds();
    /* GCOVR_EXCL_START: monotonic clock failure is an OS failure path. */
    if (now < 0) {
      (void)fprintf(stderr, "Could not read monotonic clock: %s\n",
                    strerror(errno));
      goto done;
    }
    /* GCOVR_EXCL_STOP */
    if (now >= sender.timer_deadline_ms) {
      gbn_sender_on_timeout(&sender, now, &actions);
      if (actions.failed) {
        (void)fprintf(stderr,
                      "Transfer failed after 10 consecutive timeouts without "
                      "ACK progress.\n");
        goto done;
      }
      /* GCOVR_EXCL_START: packet encoding/send failure comes from
         library/OS. */
      if (!send_protocol_packets(fd, &actions)) {
        goto done;
      }
      /* GCOVR_EXCL_STOP */
      continue;
    }

    {
      int64_t remaining = sender.timer_deadline_ms - now;
      ready = wait_readable(
          fd, remaining > (int64_t)INT_MAX ? INT_MAX : (int)remaining);
    }
    /* GCOVR_EXCL_START: poll failure is an OS failure path. */
    if (ready < 0) {
      (void)fprintf(stderr, "Network error while waiting for ACK: %s\n",
                    strerror(errno));
      goto done;
    }
    /* GCOVR_EXCL_STOP */
    if (ready == 0) {
      continue;
    }

    {
      unsigned char bytes[PACKET_MAX_SIZE + 1U];
      transfer_packet ack;
      ssize_t received = recv(fd, bytes, sizeof(bytes), 0);

      /* GCOVR_EXCL_START: recv failures are OS failure paths. */
      if (received < 0) {
        if (errno == EINTR) {
          continue;
        }
        (void)fprintf(stderr, "Network error while receiving ACK: %s\n",
                      strerror(errno));
        goto done;
      }
      /* GCOVR_EXCL_STOP */
      if ((size_t)received > PACKET_MAX_SIZE ||
          !packet_decode(bytes, (size_t)received, &ack) ||
          ack.type != PACKET_ACK) {
        continue;
      }
      now = monotonic_milliseconds();
      /* GCOVR_EXCL_START: monotonic clock failure is an OS failure path. */
      if (now < 0) {
        (void)fprintf(stderr, "Could not read monotonic clock: %s\n",
                      strerror(errno));
        goto done;
      }
      /* GCOVR_EXCL_STOP */
      gbn_sender_on_ack(&sender, &ack, now, &actions);
      /* GCOVR_EXCL_START: packet encoding/send failure comes from
         library/OS. */
      if (!send_protocol_packets(fd, &actions)) {
        goto done;
      }
      /* GCOVR_EXCL_STOP */
    }
  }
  success = sender.completed;

done:
  /* GCOVR_EXCL_START: close failure is an OS failure path. */
  if (close(fd) != 0) {
    (void)fprintf(stderr, "Could not close relay socket: %s\n",
                  strerror(errno));
    success = 0;
  }
  /* GCOVR_EXCL_STOP */
  free(packets.packets);
  return success ? 0 : 2;
}

int receive_transfer(const cli_options *options)
{
  FILE *output;
  int fd;
  int64_t now;
  gbn_receiver receiver;
  int success;

  output = fopen(options->file, "wb");
  if (output == NULL) {
    (void)fprintf(stderr, "Could not open output file '%s': %s\n",
                  options->file, strerror(errno));
    return 2;
  }

  fd = open_relay_socket(options);
  if (fd < 0) {
    (void)fclose(output);
    return 2;
  }
  if (!register_receiver(fd, options->session)) {
    (void)close(fd);
    (void)fclose(output);
    return 2;
  }

  now = monotonic_milliseconds();
  /* GCOVR_EXCL_START: monotonic clock failure is an OS failure path. */
  if (now < 0) {
    (void)fprintf(stderr, "Could not read monotonic clock: %s\n",
                  strerror(errno));
    (void)close(fd);
    (void)fclose(output);
    return 2;
  }
  /* GCOVR_EXCL_STOP */
  gbn_receiver_init(&receiver, now);
  success = receive_packets(fd, &output, &receiver);
  /* GCOVR_EXCL_START: fclose failure is a standard-library failure path. */
  if (output != NULL && fclose(output) != 0) {
    (void)fprintf(stderr, "Could not close received file: %s\n",
                  strerror(errno));
    success = 0;
  }
  /* GCOVR_EXCL_STOP */
  /* GCOVR_EXCL_START: close failure is an OS failure path. */
  if (close(fd) != 0) {
    (void)fprintf(stderr, "Could not close relay socket: %s\n",
                  strerror(errno));
    success = 0;
  }
  /* GCOVR_EXCL_STOP */
  return success ? 0 : 2;
}

static int start_mock_relay(int *socket_fd, pid_t *child, unsigned int *port,
                            int mode) {
  struct sockaddr_in address;
  socklen_t address_size = (socklen_t)sizeof(address);
  int fd = socket(AF_INET, SOCK_DGRAM, 0);
  if (fd < 0) {
    return 0;
  }
  memset(&address, 0, sizeof(address));
  address.sin_family = AF_INET;
  address.sin_addr.s_addr = htonl(INADDR_LOOPBACK);
  address.sin_port = htons(0U);
  if (bind(fd, (struct sockaddr *)&address, sizeof(address)) != 0 ||
      getsockname(fd, (struct sockaddr *)&address, &address_size) != 0) {
    close(fd);
    return 0;
  }
  *port = (unsigned int)ntohs(address.sin_port);
  *child = fork();
  if (*child < 0) {
    close(fd);
    return 0;
  }
  if (*child == 0) {
    int result = mock_relay_run(fd, mode);
    close(fd);
    _exit(result);
  }
  *socket_fd = fd;
  return 1;
}

static int write_test_file(const char *path, const unsigned char *bytes,
                           size_t size) {
  int fd = mkstemp((char *)path);
  size_t written = 0U;
  if (fd < 0) {
    return 0;
  }
  while (written < size) {
    ssize_t result = write(fd, bytes + written, size - written);
    if (result <= 0) {
      close(fd);
      return 0;
    }
    written += (size_t)result;
  }
  return close(fd) == 0;
}

  }
  return WEXITSTATUS(status) == 0;
}

static void set_wire_checksum(unsigned char *bytes, size_t size) {
  uint16_t checksum;
  bytes[2] = 0U;
  bytes[3] = 0U;
  checksum = packet_checksum(bytes, size);
  bytes[2] = (unsigned char)(checksum >> 8U);
  bytes[3] = (unsigned char)(checksum & 0xffU);
}

static void make_data_packets(transfer_packet *packets, uint32_t count) {
  uint32_t index;
  for (index = 0U; index < count; ++index) {
    memset(&packets[index], 0, sizeof(packets[index]));
    packets[index].type = PACKET_DATA;
    packets[index].seq = index;
    packets[index].length = 1U;
    packets[index].payload[0] = (unsigned char)('a' + index);
  }
}

#define TEST_CHANNEL_CAPACITY 4096U

typedef struct {
  unsigned char bytes[PACKET_MAX_SIZE];
  size_t size;
} test_datagram;

typedef struct {
  test_datagram *datagrams;
  size_t count;
  size_t capacity;
  unsigned int dropped;
  unsigned int corrupted;
  unsigned int duplicated;
} test_channel;

static uint32_t test_random_next(uint32_t *state) {
  uint32_t value = *state;
  value ^= value << 13U;
  value ^= value >> 17U;
  value ^= value << 5U;
  *state = value;
  return value;
}

static int test_channel_push(test_channel *channel,
                             const unsigned char *bytes, size_t size) {
  test_datagram *datagram;
  if (channel->count >= channel->capacity || size > PACKET_MAX_SIZE) {
    return 0;
  }
  datagram = &channel->datagrams[channel->count++];
  memcpy(datagram->bytes, bytes, size);
  datagram->size = size;
  return 1;
}

static int test_channel_send(test_channel *channel,
                             const transfer_packet *packet,
                             uint32_t *random_state) {
  unsigned char bytes[PACKET_MAX_SIZE];
  size_t size;
  uint32_t damage;

  if (!packet_encode(packet, bytes, sizeof(bytes), &size)) {
    return 0;
  }
  damage = test_random_next(random_state) % 100U;
  if (damage < 20U) {
    ++channel->dropped;
    return 1;
  }
  if (damage < 40U) {
    uint32_t bit = test_random_next(random_state) % (uint32_t)(size * 8U);
    bytes[bit / 8U] ^= (unsigned char)(1U << (bit % 8U));
    ++channel->corrupted;
    return test_channel_push(channel, bytes, size);
  }
  if (damage < 60U) {
    ++channel->duplicated;
    return test_channel_push(channel, bytes, size) &&
           test_channel_push(channel, bytes, size);
  }
  return test_channel_push(channel, bytes, size);
}

static int test_emit_actions(test_channel *channel,
                             const gbn_actions *actions,
                             uint32_t *random_state) {
  size_t index;
  for (index = 0U; index < actions->packet_count; ++index) {
    if (!test_channel_send(channel, &actions->packets[index], random_state)) {
      return 0;
    }
  }
  return 1;
}

static int run_damaged_state_machine_transfer(uint32_t seed,
                                              unsigned int *damage_flags) {
  const size_t source_size = 7001U;
  const uint32_t packet_count =
      (uint32_t)((source_size + PACKET_MAX_PAYLOAD - 1U) /
                 PACKET_MAX_PAYLOAD);
  unsigned char *source = malloc(source_size);
  unsigned char *delivered = malloc(source_size);
  transfer_packet *packets =
      calloc(packet_count, sizeof(*packets));
  test_channel data_channel = {0};
  test_channel ack_channel = {0};
  gbn_sender sender;
  gbn_receiver receiver;
  gbn_actions actions;
  uint32_t random_state = seed;
  uint32_t index;
  size_t data_read = 0U;
  size_t ack_read = 0U;
  size_t delivered_size = 0U;
  int64_t now = 0;
  unsigned int steps = 0U;
  int success = 0;

  data_channel.capacity = TEST_CHANNEL_CAPACITY;
  ack_channel.capacity = TEST_CHANNEL_CAPACITY;
  data_channel.datagrams = calloc(data_channel.capacity,
                                  sizeof(*data_channel.datagrams));
  ack_channel.datagrams = calloc(ack_channel.capacity,
                                 sizeof(*ack_channel.datagrams));
  if (source == NULL || delivered == NULL || packets == NULL ||
      data_channel.datagrams == NULL || ack_channel.datagrams == NULL) {
    goto cleanup;
  }

  for (index = 0U; index < source_size; ++index) {
    source[index] = (unsigned char)((index * 37U + index / 11U) & 0xffU);
  }
  for (index = 0U; index < packet_count; ++index) {
    size_t offset = (size_t)index * PACKET_MAX_PAYLOAD;
    size_t remaining = source_size - offset;
    size_t length = remaining < PACKET_MAX_PAYLOAD
                        ? remaining
                        : PACKET_MAX_PAYLOAD;
    packets[index].type = PACKET_DATA;
    packets[index].seq = index;
    packets[index].length = (uint16_t)length;
    memcpy(packets[index].payload, source + offset, length);
  }

  if (!gbn_sender_init(&sender, packets, packet_count, 5U, 40) ) {
    goto cleanup;
  }
  gbn_receiver_init(&receiver, now);
  gbn_sender_start(&sender, now, &actions);
  if (!test_emit_actions(&data_channel, &actions, &random_state)) {
    goto cleanup;
  }

  while (!sender.completed && !sender.failed && steps < 100000U) {
    ++steps;
    if (data_read < data_channel.count) {
      test_datagram datagram = data_channel.datagrams[data_read++];
      transfer_packet packet;
      if (packet_decode(datagram.bytes, datagram.size, &packet)) {
        gbn_receiver_on_packet(&receiver, &packet, now, &actions);
        if (actions.deliver_payload) {
          if (delivered_size + actions.payload_length > source_size) {
            goto cleanup;
          }
          memcpy(delivered + delivered_size, actions.payload,
                 actions.payload_length);
          delivered_size += actions.payload_length;
        }
        if (!test_emit_actions(&ack_channel, &actions, &random_state)) {
          goto cleanup;
        }
      }
      continue;
    }

    if (ack_read < ack_channel.count) {
      test_datagram datagram = ack_channel.datagrams[ack_read++];
      transfer_packet ack;
      if (packet_decode(datagram.bytes, datagram.size, &ack) &&
          ack.type == PACKET_ACK) {
        gbn_sender_on_ack(&sender, &ack, now, &actions);
        if (!test_emit_actions(&data_channel, &actions, &random_state)) {
          goto cleanup;
        }
      }
      continue;
    }

    if (!sender.timer_running) {
      goto cleanup;
    }
    now = sender.timer_deadline_ms;
    gbn_sender_on_timeout(&sender, now, &actions);
    if (actions.failed ||
        !test_emit_actions(&data_channel, &actions, &random_state)) {
      goto cleanup;
    }
  }

  success = sender.completed && !sender.failed &&
            delivered_size == source_size &&
            memcmp(source, delivered, source_size) == 0;

cleanup:
  if (damage_flags != NULL) {
    if (data_channel.dropped > 0U) {
      *damage_flags |= 1U << 0U;
    }
    if (data_channel.corrupted > 0U) {
      *damage_flags |= 1U << 1U;
    }
    if (data_channel.duplicated > 0U) {
      *damage_flags |= 1U << 2U;
    }
    if (ack_channel.dropped > 0U) {
      *damage_flags |= 1U << 3U;
    }
    if (ack_channel.corrupted > 0U) {
      *damage_flags |= 1U << 4U;
    }
    if (ack_channel.duplicated > 0U) {
      *damage_flags |= 1U << 5U;
    }
  }
  free(source);
  free(delivered);
  free(packets);
  free(data_channel.datagrams);
  free(ack_channel.datagrams);
  return success;
}

void test_sender_window_cumulative_ack_and_fin(void) {
  transfer_packet data[5];
  transfer_packet ack;
  gbn_sender sender;
  gbn_actions actions;
